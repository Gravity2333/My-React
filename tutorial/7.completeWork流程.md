# completeWork流程 
comleeleWork对应render中 归的阶段。 当beginWork走到叶子节点时，会向上回溯。

completeWork的实现你可以在 [/lib/react-reconciler/completeWork.ts](../lib/react-reconciler/completeWork.ts)中找到

completeWork做的事情主要包含

如果是HostComponent节点

1. 对于更新节点，给其flag打上Update标记，更新Props
2. 虽然beginWork中已经markRef了，但是在completeWork中也检查一次Ref
3. 对于初次挂载节点，需要创建其真实Dom节点，并且赋给fiber.stateNode, 注意 此时不会挂载，挂载统一到commit阶段
4. 完成创建的Dom之间父子元素的链接
5. 将flags，lane冒泡，合并其到subTreeFlags和childLanes上

如果是HostText节点
1. 如果是更新，则标记Update，在commit阶段会更新其content
2. 如果是挂载，则创建文本节点并且赋给stateNode
3. 冒泡flags和lanes

如果是其他类型节点
1. 冒泡其flags和lane

实现如下:
```typescript
/** 归的过程 主要逻辑有
 * 1. 不能复用的DOM创建 赋给stateNode
 * 2. 连接父子节点
 * 3. flags冒泡
 */
export function completeWork(wip: FiberNode) {
  /** 真正需要操作的 只有HostComponent和HostText */
  const pendingProps = wip.pendingProps; //需要处理props
  const currentFiber = wip.alternate; // 当前生效的fiber

  switch (wip.tag) {
    case HostComponent:
      /** 处理HostComponent的情况 */
      if (currentFiber && currentFiber.stateNode) {
        // update
        if (currentFiber.ref !== wip.ref) {
          wip.flags |= Ref;
				}
        // 检查pendingProps和memroizedProps 如何不同则打上Update更新标签
        // if (pendingProps !== wip.memorizedProps) {
        //   wip.flags |= Update;
        // }  
        // 不要这样写 会导致事件处理函数的闭包陷阱 我们需要在每次更新的时候 update新的event
        wip.flags |= Update;
      } else {
        // mount
        // 挂载阶段，直接创建DOM,保存到stateNode
        const domInstance = document.createElement(wip.type as string);
        // 在DOM上更新属性
        updateFiberProps(domInstance, pendingProps);
        // stateNode保存instance
        wip.stateNode = domInstance;
        // 把所有的children dom元素加入到instance中
        // completeWork时 其所有子节点已经完成了递归
        appendAllChildren(domInstance, wip); // 这个操作只有HostComponent处理 HostText由于已经没有子节点 不需要这样操作
      }
      // 冒泡处理属性
      bubbleProperties(wip);
      return null;
    case HostText:
      if (currentFiber && currentFiber.stateNode) {
        // update
        if (currentFiber.memorizedProps?.content !== pendingProps?.content) {
          wip.flags |= Update;
        }
      } else {
        // mount
        const textInstance = document.createTextNode(pendingProps?.content);
        wip.stateNode = textInstance;
      }
      bubbleProperties(wip);
      return null;
    case HostRoot:
    case FunctionComponent:
    case Fragment:
    case MemoComponent:
      bubbleProperties(wip);
      return null;
    default:
      console.warn("未处理的completeWork类型！");
  }
}
```

## 链接父子节点 appendAllChildren
每次创建完一个新的节点时，会将其和其字节点进行链接，连接方式为:

domInstance.appendChild(childInstance)

难点在找到子节点，你也许会说，可以顺着 fiber.child -> sibling -> sibling 找！是的 但是你要考虑到，如果某个子节点是“虚拟节点” 也就是如 FunctionComponent Fragment 等等，其没有真实DOM结构与其相对应，这就需要继续寻找子节点

需要深度优先寻找其一层Host节点，找到Host节点就归，不需要继续往下找了
寻找步骤大致为
```typescript
let node = wip.child // 找到wip的第一个孩子 初始化
while(1){
    if(node.tag === HostComponent || node.tag === HostText){
        // 找到Host元素了，此时append元素，并且递结束，开始回溯
        currentInstance.appendChild(node.stateNode)
    }else if(node.child){
        // 当前Fiber是虚拟元素，找当前Fiber的子Fiber，返回继续判断
        node = node.child
        continue
    }

    // 执行到此，表示当前分支已经处理完成，要么已经完成插入，要么已经找到叶子节点

    // 开始找当前节点的兄弟节点，因为要把所有的一层Host节点都加入到currentIndex的子节点

    while(node.sibling !== null){
        if(node.return === wip || node.return === null)return // 如果回溯到wip即添加的Fiber节点 或者 HostRoot节点 停止 说明没有需要插入的兄弟节点了
        node = node.return
    }

    // 执行到此 表示找到下一个等待插入的节点
    node = node.sibling
}
```

具体代码实现如下:
```typescript
/**
 * 注意 这个函数只找一层HostComponent / HostText，并且完成连接
 * 如果中间遇到Function Fragment等 会跳过去连接 直到空或者遇到的第一个Host元素为止
 *  因为挂载DOM只会挂载DOM 需要跳过Function Fragment等
 */
function appendAllChildren(instance: Element, wip: FiberNode) {
  let node = wip.child;
  while (node !== null && node !== wip) {
    if (node.tag === HostComponent || node.tag === HostText) {
      instance.appendChild(node.stateNode);
    } else if (node.child) {
      node.child.return = node; // 冗余操作
      /** 查找其子元素 */
      node = node.child;
      continue; // 返回继续检查是否为Host
    }

    /** 走到这里，说明完成这条路线的一个Host元素的append */

    /** 开始向上归 此时路径上 除了wip 和当前节点（如果不为空的情况下）其余的应该都是非Host节点 比如Function或Fragment */
    /** 如果归的时候 某个节点有sibling 那么这个sibling下的第一个Host元素 也是挂在instance下的 需要处理 */
    while (node.sibling === null) {
      // 由于wip的sibling也可能为null node.sibling === null不能判断是否到了wip 这样会一直循环下去 需要在每个循环单独判断一下，当前节点的return是不是wip
      if (node.return === wip || node.return === null) {
        return;
      }
      // 如果没有兄弟 就向上归
      node = node.return;
    }

    /** 如果有兄弟，node指向兄弟 继续循环处理 */
    node.sibling.return = node.return; // 冗余操作
    node = node.sibling;
  }
}
```

## bubbleProperties 冒泡
这个函数就是个从当前处理到的节点冒泡到跟节点的过程 如下:
```typescript
/** 把当前节点所有子节点的属性都merge到当前节点
 * 需要处理
 * 1. subtreeFlags
 * 2. childLanes
 */
function bubbleProperties(wip: FiberNode) {
  let subtreeFlags = NoFlags;
  let childLanes = NoLane;
  let node = wip.child || null;
  while (node !== null) {
    // merge subtreeFlags
    subtreeFlags |= node.subTreeFlags;
    subtreeFlags |= node.flags;
    // merge childLanes
    childLanes |= node.lanes;
    childLanes |= node.childLanes;

    // 寻找下一个子节点
    node.return = wip; // 冗余操作
    node = node.sibling; // 找下一个node
  }

  wip.subTreeFlags = subtreeFlags;
  wip.childLanes = childLanes;
}
```
