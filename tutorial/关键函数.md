#### completeWork 阶段 - bubbleProperties

这里你需要区分 bubbleProperties & markUpdateLaneFromFiberToRoot
虽然看似都是冒泡，但是，bubbleProperties 是一次调用冒泡一层，把所有子节点的 flag subTreeFlags lane ChildLanes 都冒泡到父节点
markUpdateLaneFromFiberToRoot 看名字就知道，只冒泡 lane，并且，只把某个 lane 冒泡到 root
markUpdateLaneFromFiberToRoot 作用的是 current Fiber Tree 而 bubbleProperties 是新渲染的 render Fiber Tree

```typescript
function bubbleProperties(wip: FiberNode) {

    wip.subTreeFlags |= wip.subTreeFlags;

    wip.childLanes |= wip.lanes;
    wip.childLanes |= wip.childLanes;
```

```typescript
class Update{
    action: any,
    next: Update| null,
    lane: Lane,
    constructor(action: any,lane: Lane){
        this.action = action
        this.lane = lane
        this.next = null
    }
}

class updateQueue{
    shared: {
        pedning: null,
    }

    dispatch: null,
    baseState: State,
    baseQueue: Update,
    hasEagerState:boolean
    eagerState: any
    lastRenderedState: State
    consturctor(){
        this.shared.pending  = null
        this.baseState = null
        this.baseQueue = null
        this.lastRenderedState = null
        this.hasEagerState=false
        this.eagerState = null
    }


    enqueue(update: Update,fiber: FiberNode,lane: Lane){
        if(!this.shared.pending){
            this.shared.pending = update
            update.next = update
        }else{
            update.next = this.shared.pending.next
            this.shared.pending.next = update
            this.shared.pending = update
        }

        // fiber.lane 和 fiber.alternate.lane 都要mergeLane
        fiber.lane = mergeLane(fiber.lane,lane)
        if(fiber.alternate){
           fiber.alternate.lane = mergeLane(fiber.alternate.lane,lane)
        }
    }

    process(lane: Lane,onSkipUpdate?: (update:Update) => void){
        // 暂停时的状态
        let baseState = this.baseState
        let baseQueue = this.baseQueue

        // 连接baseQueue和shared.pending
        if(this.shared.pending !== null && baseState !== null){
            /** 实现 */
            const baseFirst = baseQueue.next
            const pendingFirst = this.shared.pending.next

            baseQueue.next=pendingFirst
            this.shared.pending.next = baseFirst
            baseQueue = baseFirst
        }

        let newBaseState = baseState
        let memorizedState = newBaseState
        let newBaseQueueFirst = null
        let newBaseQueueLast = null

        let baseQueueItem = baseQueue.next
        while(baseQueueItem!==baseQueue){
            const baseQueueLane = baseQueueItem.lane
            if(isSubsetOfLanes(baseQueueLane,lane)){
                // 有权限
                if(newBaseQueueFirst !== null){
                    const clone = new Update(baseQueueItem.action,NoLane)
                    newBaseQueueLast = newBaseQueueLast.next = clone
                }
                if(baseQueueItem.hasEagerState){
                    memorizedState = baseQueueItem.eagerState
                }else{
                    if(typeof baseQueueItem.action === 'function' ){
                         memorizedState = baseQueueItem.action()
                    }else{
                        memorizedState = baseQueueItem.action,
                    }
                }
            }else{
                // 无权限
                const clone = new Update(baseQueueItem.action,NoLane)
                if(newBaseQueueFirst === null){
                    // 此处暂停
                    newBaseQueueLast = newBaseFirst = clone
                    newBaseState = memorizedState
                }else{
                    newBaseQueueLast = newBaseQueueLast.next = clone
                }
            }
        }

        if(newBaseQueueFirst === null){
            baseState = memorizedState
        }else{
            // 连接环
            newBaseQueue = newBaseLast.next = newBaseFirst
        }
        this.baseQueue = newBaseQueue
        this.baseState = newBaseState
        return {memorizedState}
    }
}
```

```typescript
function renderWithHooks(
    wip: FiberNode, // 当前处理的fiber
    Component: any, // 当前函数组件的函数本身
    lane: Lane, // 优先级
){
    renderLane = lane
    currentRenderingFiber = wip
    // 重制 wip.memorizedState & updateQueue
    wip.memorizedState = null
    wip.updateQueue = null

    const current = wip.alternate
    /** 设置hooks上下文 */
    if(!current){
        // 第一次渲染
        currentDispatcher.current = {
            useState: mountState,
            useMemo: mountMemo,
            useEffect: mountEffect
            ...
        }
    }else{
        // 更新
        currentDispatcher.current = {
            useState: updateState,
            useMemo: updateMemo,
            useEffect: updateEffect
            ...
        }
    }

    // 运行函数
    const elem = Component(wip.pendingProps)

    // 恢复现场
    currentRenderingFiber = null
    renderLane = NoLane
    currentHook = null
    workInProgressHook = null
    currentDispatcher.current = null
    // 返回elem
    return elem
}
```

#### mountWorkInProgressHook & updateWorkInProgressHook

在挂载/更新阶段，获取当前的 hook

```typescript
function mountWorkInProgressHook() {
  if (!currentRenderingFiber) throw new Error("不能在函数组件外部调用hooks");
  const hook: Hook = {
    updateQueue: new UpdateQueue(),
    memorizedState: null,
    next: null,
  };

  if (workInProgressHook === null) {
    // 首次挂载
    workInProgressHook = currentRenderingFiber.memorizedState = hook;
  } else {
    // 二次挂载
    workInProgressHook = workInProgressHook.next = hook;
  }

  return hook;
}

function updateWorkInProgressHook() {
  if (!currentRenderingFiber) throw new Error("不能在函数组件外部调用hooks");
  const currentFiber = currentRenderingFiber.alternate;
  if (currentHook === null) {
    // 第一次更新
    currentHook = currentFiber.memorizedState;
  } else {
    currentHook = currentHook.next;
  }

  if (!currentHook) {
    throw new Error("rendering more hooks than previouse render!");
  }

  // 根据currentHook 构建新的hook
  const hook: Hook = {
    updateQueue: currentHook.updateQueue,
    memorizedState: currentHook.memorizedState,
    next: null,
  };

  // 挂载新的hook
  if (workInProgressHook === null) {
    // 首次挂载
    workInProgressHook = currentRenderingFiber.memorizedState = hook;
  } else {
    // 二次挂载
    workInProgressHook = workInProgressHook.next = hook;
  }

  return hook;
}
```

```typescript
function dispatchSetState(
  wip: FiberNode,
  updateQueue: UpdateQueue, // hook的updateQueue
  action: Action
) {
  const lane = requestUpdateLane();
  const update = new Update(action, lane);

  // eagerState
  if (updateQueue.shared.pending === null) {
    const result;
    if (typeof action === "function") {
      result = action();
    } else {
      result = action;
    }

    if (Object.is(updateQueue.lastRenderedState, result)) {
      // 相同 eagerState
      update.hasEagerState = true;
      update.eagerState = result;
    }
  }
  // 入队
  updateQueue.enqueue(update, wip, lane);
  // 调度
  scheduleUpdateOnFiber(wip, lane);
}
```

```typescript
function pushEffect(tags: EffectTag, create: any, destory: any, deps: Deps) {
  const effect: Efffect = {
    tags,
    create,
    destory,
    deps: deps === undefined ? null : deps,
    next: null,
  };

  if (!currentRenderingFiber.updateQueue) {
    const fcUpdateQueue = new FCUpdateQueue();
    effect.next = effect; // 自环
    fcUpdateQueue.lastEffect = effect;
    currentRenderingFiber.updateQueue = fcUpdateQueue;
  } else {
    const fcUpdateQueue = currentRenderingFiber.updateQueue;
    effect.next = fcUpdateQueue.lastEffect.next;
    fcUpdateQueue.lastEffect.next = effect;
  }
  return effect;
}
```
